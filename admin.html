Cloud Database & Authentication

Connect Supabase or Firebase to manage real-time client leads and secure admin authentication in the cloud.

Supabase SQL Table Schema & Zero-Leak RLS 
-- Create inquiries table in Supabase SQL editor:
create table if not exists inquiries (
  id uuid default gen_random_uuid() primary key,
  name text not null,
  email text not null,
  service text,
  message text,
  status text default 'New',
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- Enable Row Level Security (BLOCKS ALL READS BY DEFAULT):
alter table inquiries enable row level security;

-- 1. Public leads submission (INSERT ONLY - zero read permissions for anon):
create policy "Allow insert for visitors" on inquiries 
  for insert to anon, authenticated 
  with check (true);

-- 2. Authenticated studio owner access (FULL READ/WRITE/DELETE):
create policy "Allow authenticated admin full access" on inquiries 
  for all to authenticated 
  using (auth.role() = 'authenticated') 
  with check (auth.role() = 'authenticated');
Firebase Firestore Security Rules 
// Paste into Firebase Firestore Rules console:
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /inquiries/{inquiryId} {
      allow create: if true;
      allow read, update, delete: if request.auth != null;
    }
  }
}
